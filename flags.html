regionNames = new Intl.DisplayNames(['en'], { type: 'region' });

function getCountryName(code) {
  if (!code) return "";
  try {
    return regionNames.of(code.toUpperCase()) || code.toUpperCase();
  } catch (e) {
    return code.toUpperCase();
  }
}

// Calculate match percentage (distance 0 = 100%, distance >= 40 = 0%)
function getMatchPct(dist) {
  return Math.max(0, Math.min(100, Math.round(100 - (dist / 40) * 100)));
}

// Quiz Component State
quizContainer = {
  const container = html`<div style="font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, sans-serif; max-width: 650px;"></div>`;
  
  let score = 0;
  let streak = 0;
  let answered = false;
  
  // Pool of interesting countries with clear twins
  const challengePool = ["ro", "id", "us", "fr", "ie", "au", "cl", "co", "nl", "de", "sn", "eg", "jp"];

  function nextQuestion() {
    answered = false;
    
    // Pick random target
    const targetCode = challengePool[Math.floor(Math.random() * challengePool.length)];
    const targetName = getCountryName(targetCode);
    const targetStats = statsMap.get(targetCode);
    const correctCode = targetStats.closest_match;
    const correctName = getCountryName(correctCode);

    // Pick 3 random distractors
    const distractors = [];
    while (distractors.length < 3) {
      const rand = flag_codes[Math.floor(Math.random() * flag_codes.length)];
      if (rand !== targetCode && rand !== correctCode && !distractors.includes(rand)) {
        distractors.push(rand);
      }
    }

    const options = [correctCode, ...distractors].sort(() => Math.random() - 0.5);

    container.innerHTML = `
      <div style="background: #ffffff; border: 1px solid #e2e8f0; border-radius: 12px; padding: 24px; box-shadow: 0 4px 10px rgba(0,0,0,0.06);">
        <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 16px;">
          <span style="font-weight: 700; color: #475569; font-size: 0.95rem;">Doppelgänger Challenge</span>
          <div style="display: flex; gap: 10px;">
            <span style="background: #e2e8f0; padding: 4px 10px; border-radius: 20px; font-weight: 700; font-size: 0.85rem; color: #334155;">Score: ${score}</span>
            <span style="background: #fef3c7; padding: 4px 10px; border-radius: 20px; font-weight: 700; font-size: 0.85rem; color: #b45309;">🔥 Streak: ${streak}</span>
          </div>
        </div>

        <p style="font-size: 1.05rem; margin-bottom: 12px; color: #0f172a;">
          Which flag is the <strong>closest visual twin</strong> to <strong>${targetName}</strong>?
        </p>

        <div style="text-align: center; margin: 15px 0 25px 0;">
          <img 
            src="data/${targetCode}.png"
            onerror="this.src='https://flagcdn.com/w640/${targetCode}.png'" 
            style="width: 220px; height: 140px; object-fit: contain; border: 2px solid #3b82f6; border-radius: 8px; box-shadow: 0 4px 8px rgba(0,0,0,0.1);"
          />
        </div>

        <div id="options-grid" style="display: grid; grid-template-columns: 1fr 1fr; gap: 14px;">
          ${options.map(opt => `
            <button 
              data-code="${opt}"
              class="quiz-btn"
              style="display: flex; flex-direction: column; align-items: center; gap: 8px; padding: 12px; border: 2px solid #e2e8f0; border-radius: 8px; background: #ffffff; cursor: pointer; transition: all 0.15s;"
            >
              <img 
                src="data/${opt}.png" 
                onerror="this.src='https://flagcdn.com/w320/${opt}.png'" 
                style="width: 120px; height: 75px; object-fit: contain; border: 1px solid #cbd5e1; border-radius: 4px; pointer-events: none;"
              />
              <span style="font-weight: 600; font-size: 0.85rem; color: #1e293b; pointer-events: none;">${getCountryName(opt)}</span>
            </button>
          `).join("")}
        </div>

        <div id="feedback-area" style="margin-top: 18px; min-height: 40px; text-align: center;"></div>
      </div>
    `;

    // Attach click listeners to options
    container.querySelectorAll(".quiz-btn").forEach(btn => {
      btn.addEventListener("click", () => {
        if (answered) return;
        answered = true;
        const chosen = btn.getAttribute("data-code");
        const feedback = container.querySelector("#feedback-area");

        if (chosen === correctCode) {
          score += 10;
          streak += 1;
          btn.style.borderColor = "#10b981";
          btn.style.background = "#ecfdf5";
          feedback.innerHTML = `
            <div style="color: #059669; font-weight: 700; font-size: 1.1rem; margin-bottom: 10px;">
              🎉 Correct! ${correctName} is the closest match!
            </div>
            <button id="next-btn" class="btn btn-primary" style="padding: 8px 20px; font-weight: 600;">Next Flag ➔</button>
          `;
        } else {
          streak = 0;
          btn.style.borderColor = "#ef4444";
          btn.style.background = "#fef2f2";
          
          // Highlight correct one
          container.querySelectorAll(".quiz-btn").forEach(b => {
            if (b.getAttribute("data-code") === correctCode) {
              b.style.borderColor = "#10b981";
              b.style.background = "#ecfdf5";
            }
          });

          feedback.innerHTML = `
            <div style="color: #dc2626; font-weight: 700; font-size: 1.05rem; margin-bottom: 10px;">
              Not quite! The closest twin was ${correctName}.
            </div>
            <button id="next-btn" class="btn btn-secondary" style="padding: 8px 20px; font-weight: 600;">Try Another ➔</button>
          `;
        }

        container.querySelector("#next-btn").addEventListener("click", nextQuestion);
      });
    });
  }

  nextQuestion();
  return container;
}
